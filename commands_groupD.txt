with open('ASV_taxonomy.tsv', 'r') as fin, open('taxonomy_formatted.tsv', 'w') as fout:
    fout.write("Feature ID\tTaxon\tConfidence\n")
    next(fin)
    for line in fin:
        parts = line.strip().split("\t")
        if len(parts) < 2:
            continue
        asv_id     = parts[0]
        taxonomy   = "; ".join(p for p in parts[1:7] if p.strip())
        confidence = parts[7] if len(parts) > 7 else "1.0"
        if not taxonomy or "Eukaryota" in taxonomy:
            continue
        fout.write(f"{asv_id}\t{taxonomy}\t{confidence}\n")
        
qiime tools import \
  --type 'FeatureData[Taxonomy]' \
  --input-path taxonomy_formatted.tsv \
  --input-format TSVTaxonomyFormat \
  --output-path taxonomy.qza
